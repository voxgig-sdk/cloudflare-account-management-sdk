Tools

cloudflare-account-management_list

List records for an entity.

{
  "type": "object",
  "required": [
    "entity"
  ],
  "properties": {
    "entity": {
      "type": "string",
      "description": "account | account_setting | cache_purge_api_response_single_id | cps_communication_preference | iam_api_response_single_id | iam_response_single_account | iam_single_user | kamino_environment | resource | resource_tagging | resource_tagging_tag_key_summary_response_collection | resource_tagging_tag_keys_response_collection | resource_tagging_tag_values_response_collection | resource_tagging_tagged_resource_response_single | zone | zone_hold"
    },
    "query": {
      "type": "object",
      "description": "Optional match fields for the selected entity."
    }
  }
}

Supported entities: account, account_setting, cps_communication_preference, iam_api_response_single_id, kamino_environment, resource, resource_tagging_tag_key_summary_response_collection, resource_tagging_tag_keys_response_collection, resource_tagging_tagged_resource_response_single, zone.
